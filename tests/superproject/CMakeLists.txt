cmake_minimum_required(VERSION 3.20)
project(rateweaver_superproject_probe LANGUAGES CXX)
foreach(dependency TimeShield nlohmann_json pugixml kurlyk RateWeaverKurlykHttp CURL)
    set(CMAKE_DISABLE_FIND_PACKAGE_${dependency} OFF)
endforeach()
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS ON)
add_library(host_time_shield INTERFACE)
add_library(time_shield::time_shield ALIAS host_time_shield)
target_include_directories(host_time_shield INTERFACE "${RATEWEAVER_SOURCE_DIR}/external/time-shield-cpp/include")
target_compile_definitions(host_time_shield INTERFACE TIME_SHIELD_ENABLE_NTP_CLIENT=1)
if(HOST_JSON)
    find_package(nlohmann_json 3.11 CONFIG REQUIRED)
endif()
if(HOST_ECB)
    find_package(pugixml 1.13 CONFIG REQUIRED)
endif()
if(HOST_KURLYK AND HOST_CURL_ONLY)
    find_package(CURL REQUIRED)
    set(FETCHCONTENT_FULLY_DISCONNECTED ON)
    set(RATEWEAVER_CURL_MODE PREBUILT CACHE STRING "" FORCE)
elseif(HOST_KURLYK)
    find_package(RateWeaverKurlykHttp 1.1.1 QUIET CONFIG)
    if(TARGET RateWeaverKurlykHttp::client)
        set(host_http_target RateWeaverKurlykHttp::client)
    else()
        find_package(kurlyk 1.1.1 CONFIG REQUIRED)
        set(host_http_target kurlyk::kurlyk)
    endif()
    add_library(host_kurlyk INTERFACE)
    if(NOT TARGET kurlyk::kurlyk)
        add_library(kurlyk::kurlyk ALIAS host_kurlyk)
    endif()
    target_link_libraries(host_kurlyk INTERFACE ${host_http_target})
    target_compile_definitions(host_kurlyk INTERFACE
        KURLYK_AUTO_INIT=0 KURLYK_WEBSOCKET_SUPPORT=0 KURLYK_AUTH_SUPPORT=0 KURLYK_OAUTH_SUPPORT=0)
    get_target_property(host_definitions host_kurlyk INTERFACE_COMPILE_DEFINITIONS)
endif()
foreach(dependency TimeShield nlohmann_json pugixml kurlyk RateWeaverKurlykHttp CURL)
    set(CMAKE_DISABLE_FIND_PACKAGE_${dependency} ON)
endforeach()
set(RATEWEAVER_INSTALL OFF CACHE BOOL "" FORCE)
# Existing targets win even when a caller explicitly requests bundled fallback.
set(RATEWEAVER_DEPENDENCY_MODE BUNDLED CACHE STRING "" FORCE)
set(RATEWEAVER_BUILD_PROVIDERS ${HOST_JSON} CACHE BOOL "" FORCE)
set(RATEWEAVER_ENABLE_ECB ${HOST_ECB} CACHE BOOL "" FORCE)
set(RATEWEAVER_ENABLE_KURLYK ${HOST_KURLYK} CACHE BOOL "" FORCE)
set(duplicate_candidates rateweaver_core_tests time_shield nlohmann_json pugixml
    rateweaver_kurlyk_http_dependency libcurl_static)
if(HOST_CURL_ONLY)
    list(REMOVE_ITEM duplicate_candidates rateweaver_kurlyk_http_dependency)
endif()
set(existing_targets)
foreach(candidate IN LISTS duplicate_candidates)
    if(TARGET "${candidate}")
        list(APPEND existing_targets "${candidate}")
    endif()
endforeach()
add_subdirectory("${RATEWEAVER_SOURCE_DIR}" rateweaver)
foreach(candidate IN LISTS duplicate_candidates)
    if(TARGET "${candidate}" AND NOT "${candidate}" IN_LIST existing_targets)
        message(FATAL_ERROR "Embedded build added duplicate dependency or standalone tests: ${candidate}")
    endif()
endforeach()
if(NOT CMAKE_CXX_STANDARD EQUAL 20 OR NOT BUILD_SHARED_LIBS)
    message(FATAL_ERROR "RateWeaver mutated host global build settings")
endif()
if(HOST_KURLYK AND NOT HOST_CURL_ONLY)
    get_target_property(after_definitions host_kurlyk INTERFACE_COMPILE_DEFINITIONS)
    if(NOT after_definitions STREQUAL host_definitions)
        message(FATAL_ERROR "RateWeaver changed host kurlyk feature definitions")
    endif()
endif()
if(HOST_CURL_ONLY AND (NOT RATEWEAVER_CURL_PROVIDER STREQUAL "HOST" OR
        EXISTS "${CMAKE_CURRENT_BINARY_DIR}/rateweaver/_deps"))
    message(FATAL_ERROR "Embedded host curl was not reused without dependency acquisition")
endif()
add_executable(embedded_consumer "${CMAKE_CURRENT_LIST_DIR}/../install_consumer/main.cpp")
target_link_libraries(embedded_consumer PRIVATE rateweaver::rateweaver)
set_target_properties(embedded_consumer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
enable_testing()
add_test(NAME embedded_consumer COMMAND embedded_consumer)
if(HOST_JSON OR HOST_ECB OR HOST_KURLYK)
    add_executable(embedded_components "${CMAKE_CURRENT_LIST_DIR}/../install_components/main.cpp")
    set_target_properties(embedded_components PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(HOST_JSON)
        target_link_libraries(embedded_components PRIVATE rateweaver::providers)
        target_compile_definitions(embedded_components PRIVATE CONSUME_JSON=1)
    endif()
    if(HOST_ECB)
        target_link_libraries(embedded_components PRIVATE rateweaver::ecb)
        target_compile_definitions(embedded_components PRIVATE CONSUME_ECB=1)
    endif()
    if(HOST_KURLYK)
        target_link_libraries(embedded_components PRIVATE rateweaver::kurlyk)
        target_compile_definitions(embedded_components PRIVATE CONSUME_KURLYK=1)
        # The host explicitly deploys its own runtime; the embedded library does not.
        rateweaver_copy_runtime_dlls(embedded_components)
    endif()
    add_test(NAME embedded_components COMMAND embedded_components)
endif()
