cmake_minimum_required(VERSION 3.21)
project(rateweaver_curl_resolution_probe LANGUAGES CXX)
include("${RATEWEAVER_SOURCE_DIR}/cmake/rateweaverDependencies.cmake")
set(CMAKE_DISABLE_FIND_PACKAGE_CURL ON)
set(FETCHCONTENT_FULLY_DISCONNECTED ON)
if(PROBE STREQUAL "host")
    add_library(host_curl INTERFACE)
    add_library(CURL::libcurl ALIAS host_curl)
    target_compile_definitions(host_curl INTERFACE HOST_CURL_SETTING=1)
    foreach(mode AUTO SYSTEM PREBUILT SOURCE BUNDLED)
        set(RATEWEAVER_CURL_MODE "${mode}")
        rateweaver_resolve_curl()
    endforeach()
    get_target_property(definitions host_curl INTERFACE_COMPILE_DEFINITIONS)
    if(NOT definitions STREQUAL "HOST_CURL_SETTING=1")
        message(FATAL_ERROR "Host curl target changed")
    endif()
    if(TARGET libcurl_static OR DEFINED FETCHCONTENT_BASE_DIR OR EXISTS "${PROJECT_BINARY_DIR}/_deps")
        message(FATAL_ERROR "Host curl reuse triggered dependency acquisition")
    endif()
elseif(PROBE STREQUAL "legacy-normal")
    add_library(host_time INTERFACE)
    add_library(time_shield::time_shield ALIAS host_time)
    set(RATEWEAVER_CURL_MODE BUNDLED)
    set(RATEWEAVER_INSTALL OFF)
    set(RATEWEAVER_BUILD_TESTS OFF)
    set(RATEWEAVER_BUILD_EXAMPLES OFF)
    add_subdirectory("${RATEWEAVER_SOURCE_DIR}" rateweaver)
    get_property(curl_mode CACHE RATEWEAVER_CURL_MODE PROPERTY VALUE)
    if(NOT curl_mode STREQUAL "SOURCE")
        message(FATAL_ERROR "Legacy curl mode was not normalized in the cache")
    endif()
elseif(PROBE STREQUAL "system-missing")
    set(RATEWEAVER_CURL_MODE SYSTEM)
    rateweaver_resolve_curl()
elseif(PROBE STREQUAL "unsupported-prebuilt")
    # Exercise the platform rejection even when the test host supports PREBUILT.
    set(MINGW OFF)
    set(RATEWEAVER_CURL_MODE PREBUILT)
    rateweaver_resolve_curl()
else()
    message(FATAL_ERROR "Unknown resolver probe")
endif()
