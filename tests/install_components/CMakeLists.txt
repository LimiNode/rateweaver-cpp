cmake_minimum_required(VERSION 3.20)
project(rateweaver_installed_components LANGUAGES CXX)
set(components)
if(CONSUME_JSON)
    list(APPEND components providers)
endif()
if(CONSUME_ECB)
    list(APPEND components ecb)
endif()
find_package(rateweaver 0.1 CONFIG REQUIRED COMPONENTS ${components})
add_executable(consumer main.cpp)
set_target_properties(consumer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(CONSUME_JSON)
    target_link_libraries(consumer PRIVATE rateweaver::providers)
    target_compile_definitions(consumer PRIVATE CONSUME_JSON=1)
endif()
if(CONSUME_ECB)
    target_link_libraries(consumer PRIVATE rateweaver::ecb)
    target_compile_definitions(consumer PRIVATE CONSUME_ECB=1)
endif()
enable_testing()
add_test(NAME consumer COMMAND consumer)
